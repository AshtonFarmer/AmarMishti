const transition = document.getElementById("memoryTransition");
const universe = document.getElementById("memoryUniverse");
const container = document.getElementById("memoryContainer");
const restart = document.getElementById("restartUniverse");
const sunsetCollapse = document.getElementById("sunsetCollapse");

function placeMemory(photoName, index) {
  const img = document.createElement("img");
  img.src = photoName;
  img.className = "floating-memory memory-pos-" + ((index % 8) + 1);
  img.alt = "A memory";
  container.appendChild(img);
}

function startSunsetRestart() {
  sunsetCollapse.classList.remove("hidden-sunset");

  if(navigator.vibrate){
    navigator.vibrate([160,80,120,80,160]);
  }

  // Let the detailed sunset orb open, shrink, heartbeat, then fade into the birthday page.
  setTimeout(() => {
    window.location.href = "index.html?fadein=1";
  }, 7600);
}

setTimeout(() => {
  transition.classList.add("fade-away-memory");

  setTimeout(() => {
    transition.style.display = "none";
    universe.classList.remove("hidden-memory");

    memoryPhotos.forEach((photo, index) => {
      setTimeout(() => {
        placeMemory(photo, index);
      }, 700 + index * 850);
    });

    setTimeout(() => {
      restart.classList.remove("hidden-memory");
    }, 15500);

  }, 1200);
}, 6200);

restart.addEventListener("click", (e) => {
  e.preventDefault();

  const curtain = document.getElementById("memoryFadeCurtain");
  const message = document.querySelector(".memory-message");
  const finalLine = document.getElementById("memoryFinal");

  restart.style.pointerEvents = "none";
  restart.style.transition = "opacity 1.2s ease, transform 1.2s ease";
  restart.style.opacity = "0";
  restart.style.transform = "translateX(-50%) translateY(18px) scale(.92)";

  // Make the whole memory universe feel like it is slowly dissolving.
  if(curtain){
    curtain.classList.remove("hidden-sunset");
    requestAnimationFrame(() => {
      curtain.classList.add("fade-in-curtain");
    });
  }

  if(message){
    message.style.transition = "opacity 2.3s ease, transform 2.3s ease, filter 2.3s ease";
    message.style.opacity = "0";
    message.style.transform = "translateY(-30px) scale(.96)";
    message.style.filter = "blur(5px)";
  }

  if(finalLine){
    finalLine.style.transition = "opacity 2.3s ease, transform 2.3s ease, filter 2.3s ease";
    finalLine.style.opacity = "0";
    finalLine.style.transform = "translateY(30px) scale(.96)";
    finalLine.style.filter = "blur(5px)";
  }

  document.querySelectorAll(".floating-memory").forEach((photo,index) => {
    photo.style.transition =
      "opacity 2.9s ease, transform 2.9s ease, filter 2.9s ease";
    photo.style.transitionDelay = (index * 0.06) + "s";
    photo.style.opacity = "0";
    photo.style.transform =
      "translateY(-160px) scale(.48) rotate(" + ((index % 2 === 0) ? "-12deg" : "12deg") + ")";
    photo.style.filter = "blur(8px) brightness(.35)";
  });

  setTimeout(() => {
    startSunsetRestart();
  }, 3000);
});

function resetTimeline(){
  if(foundText){
    foundText.classList.remove("found-show");
    foundText.classList.add("found-hidden");
  }

  document.querySelectorAll(".scan-line").forEach(line => {
    line.classList.remove("show-line");
    line.classList.add("hidden-line");
  });
}

function runTimelineTransition(options){
  const searchText = options.searchText || "Searching the multiverse...";
  const foundMessage = options.foundMessage || "Tanima Das found ❤️";
  const after = options.after || function(){};

  resetTimeline();

  timelineTitle.textContent = searchText;
  foundText.textContent = foundMessage;

  timeline.classList.remove("hidden");
  timeline.style.display = "grid";
  timeline.style.opacity = "1";

  const scanLines = document.querySelectorAll(".scan-line");

  scanLines.forEach((line, index) => {
    setTimeout(() => {
      line.classList.remove("hidden-line");
      line.classList.add("show-line");
    }, 700 + (index * 650));
  });

  setTimeout(() => {
    foundText.classList.remove("found-hidden");
    foundText.classList.add("found-show");
  }, 3900);

  setTimeout(() => {
    timeline.classList.add("hidden");
    timeline.style.display = "";
    resetTimeline();
    after();
  }, 7800);
}



startBtn.addEventListener("click", async () => {
  startBtn.classList.add("hidden-btn");
  video.controls = true;
  video.muted = false;
  video.currentTime = 0;

  try{
    await video.play();
  }catch(e){
    startBtn.classList.remove("hidden-btn");
    startBtn.textContent = "Tap again to play ❤️";
    return;
  }

  try{
    if(video.requestFullscreen){
      await video.requestFullscreen();
    }else if(video.webkitEnterFullscreen){
      video.webkitEnterFullscreen();
    }else if(video.webkitRequestFullscreen){
      await video.webkitRequestFullscreen();
    }
  }catch(e){}
});

    });

  async function loadActualCode(){
    if(!actualCode) return;

    try{
      const response = await fetch("FULL_SOURCE_CODE.txt");
      const text = await response.text();
      actualCode.textContent = text;

      actualCode.textContent = `The real source code is included in this website folder as:

FULL_SOURCE_CODE.txt

It contains the HTML, CSS, and JavaScript that made this whole birthday multiverse experience work. ❤️`;

  }
